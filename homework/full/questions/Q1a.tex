We have

\begin{align*}
    2 + 5 + 8 + \ldots + 3n - 1 &= \frac{n(3n + 1)}{2}
\end{align*}

Rewritten using summation gives us

\begin{align*}
    \sum_{i = 1}^{n}{3i - 1} = \frac{n(3n + 1)}{2}
\end{align*}

\begin{poof}
    BASE CASE:

    Let $n = 1$

    \begin{align*}
        \sum_{i = 1}^{n}{3i - 1} = \sum_{i = 1}^{1}{3(1) - 1} &= 3 - 1 \\
        &= \boxed{2}
    \end{align*}

    \begin{align*}
        \frac{n(3n + 1)}{2} = \frac{1 \cdot (3(1) + 1)}{2} &= \frac{3 + 1}{2} \\
        &= \frac{4}{2} \\
        &= \boxed{2} \\
    \end{align*}

    \begin{align*}
        2 &= 2 \\
        \therefore n &= 1 \text{ is true}
    \end{align*}

    INDUCTIVE CASE:

    Assume that

    \begin{align*}
        \sum_{i = 1}^{n}{3n - 1} = \frac{n(3n + 1)}{2}
    \end{align*}

    We must prove

    \begin{align*}
        \sum_{i = 1}^{n + 1}{3i - 1} &= \frac{(n + 1)(3(n + 1) + 1)}{2} \\
        &= \frac{(n + 1)(3n + 4)}{2}
    \end{align*}

    \begin{align*}
        \sum_{i = 1}^{n + 1}{3i - 1} &= \sum_{i = 1}^{n}{3i - 1} + \left(3(n + 1) - 1\right) \\
        &= \frac{n(3n + 1)}{2} + 3(n + 1) - 1 \\
        &= \frac{n(3n + 1)}{2} + 3n + 2 \\
        &= \frac{n(3n + 1)}{2} + \frac{6n + 4}{2} \\
        &= \frac{n(3n + 1) + (6n + 4)}{2} \\
        &= \frac{3n^2 + n + 6n + 4}{2} \\
        &= \frac{3n^2 + 7n + 4}{2} \\
        &= \frac{(n + \frac{3}{3})(n + \frac{4}{3})}{2} \\
        &= \boxed{\frac{(n + 1)(3n + 4)}{2}}
    \end{align*}

    \begin{align*}
        \frac{(n + 1)(3n + 4)}{2} &= \frac{(n + 1)(3n + 4)}{2} \checkmark
    \end{align*}
\end{poof}
