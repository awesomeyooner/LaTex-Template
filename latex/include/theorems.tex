
% Header Command
\newcommand{\header}[2]
{%
    \noindent\textbf{\underline{#1}}\textbf{:} #2
}

% Definition
\newcommand{\mydef}[1]
{%
    \header{Definition}{#1}
}

\newlength{\spaceabove}
\newlength{\spacebelow}

\setlength{\spaceabove}{20pt}
\setlength{\spacebelow}{20pt}

% DEFINITION
\newtheoremstyle{definitionstyle}
  {\spaceabove}   % Space above
  {\spacebelow}   % Space below
  {}      % Body font
  {}      % Indent
  {\bfseries}  % Head font
  {}      % Punctuation after head
  {\newline} % Space after head
  {\underline{\thmname{#1}}\thmnumber{ #2}\thmnote{ (#3)}}

\theoremstyle{definitionstyle}
\newtheorem{definition}{Definition}
